   537→  // Joystick handlers
   538→  const createJoystickHandlers = useCallback((type: 'move' | 'draw') => {
   539→    let isActive = false
   540→    let centerX = 0
   541→    let centerY = 0
   542→    let maxRadius = 50
   543→    
   544→    const handleStart = (e: React.TouchEvent | React.MouseEvent) => {
   545→      e.preventDefault()
   546→      isActive = true
   547→      const knob = type === 'move' ? moveKnobRef.current : drawKnobRef.current
   548→      const rect = knob?.parentElement?.getBoundingClientRect()
   549→      if (rect) {
   550→        centerX = rect.left + rect.width / 2
   551→        centerY = rect.top + rect.height / 2
   552→        maxRadius = rect.width / 2.2
   553→      }
   554→    }
   555→    
   556→    const handleMove = (e: React.TouchEvent | React.MouseEvent) => {
   557→      if (!isActive) return
   558→      e.preventDefault()
   559→      
   560→      const knob = type === 'move' ? moveKnobRef.current : drawKnobRef.current
   561→      if (!knob) return
   562→      
   563→      const clientX = 'touches' in e ? e.touches[0].clientX : e.clientX
   564→      const clientY = 'touches' in e ? e.touches[0].clientY : e.clientY
   565→      
   566→      const dx = clientX - centerX
   567→      const dy = clientY - centerY
   568→      const dist = Math.min(Math.sqrt(dx * dx + dy * dy), maxRadius)
   569→      const angle = Math.atan2(dy, dx)
   570→      
   571→      knob.style.transform = `translate(${Math.cos(angle) * dist}px, ${Math.sin(angle) * dist}px)`
   572→      
   573→      const speed = Math.pow(dist / maxRadius, 1.3) * 3.5
   574→      
   575→      if (type === 'move') {
   576→        targetVxMove.current = Math.cos(angle) * speed
   577→        targetVyMove.current = Math.sin(angle) * speed
   578→      } else {
   579→        if (Math.abs(dx) > Math.abs(dy)) {
   580→          targetVxDraw.current = Math.sign(dx) * speed
   581→          targetVyDraw.current = 0
   582→        } else {
   583→          targetVxDraw.current = 0
   584→          targetVyDraw.current = Math.sign(dy) * speed
   585→        }
   586→      }
   587→    }
   588→    
   589→    const handleEnd = () => {
   590→      if (!isActive) return
   591→      isActive = false
   592→      
   593→      const knob = type === 'move' ? moveKnobRef.current : drawKnobRef.current
   594→      if (knob) {
   595→        knob.style.transition = 'transform 0.6s cubic-bezier(0.34, 1.56, 0.64, 1)'
   596→        knob.style.transform = 'translate(0px, 0px)'
   597→        setTimeout(() => { if (knob) knob.style.transition = '' }, 600)
   598→      }
   599→
