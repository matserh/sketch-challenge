   547→    let isActive = false

   568→      const knob = type === 'move' ? moveKnobRef.current : drawKnobRef.current

   609→        targetVxMove.current = 0
   610→        targetVyMove.current = 0
   611→      } else {
   612→        targetVxDraw.current = 0
   613→        targetVyDraw.current = 0
   614→      }
   615→    }
   616→    
   617→    return { handleStart, handleMove, handleEnd }
